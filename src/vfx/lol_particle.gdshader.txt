shader_type spatial;
render_mode unshaded, cull_disabled, depth_draw_never, shadows_disabled, specular_disabled, __BLEND__;
// League particle material: texture x particle color, multiply texture, uv scroll, erosion dissolve,
// soft depth fade, billboard modes and flipbooks.
// additive materials sample raw (gamma) values to emulate League's gamma-space additive blending
uniform sampler2D tex : __TEXHINT__ filter_linear_mipmap, repeat_enable;
uniform sampler2D tex2 : __TEXHINT__ filter_linear_mipmap, repeat_enable;
uniform bool linear_color = false;   // alpha-blended materials: convert League's gamma colors to linear
uniform bool has_tex2 = false;
uniform sampler2D ero_tex : filter_linear_mipmap, repeat_enable;
uniform sampler2D ero_curve : repeat_disable, filter_linear;
uniform bool has_erosion = false;
uniform vec4 ero_channel = vec4(1.0, 0.0, 0.0, 0.0);
uniform float ero_feather = 0.1;
uniform sampler2D uv_curve : repeat_disable, filter_linear;   // rg = main uv offset, ba = mult uv offset (over life)
uniform bool has_uv_curve = false;
uniform sampler2D depth_tex : hint_depth_texture, filter_nearest;
uniform float soft_dist = 0.0;
uniform vec2 texdiv = vec2(1.0, 1.0);
uniform float total_frames = 1.0;
uniform int billboard = 1;          // 0 = particle transform, 1 = camera facing, 2 = axis (particle Y)
uniform vec2 uv_scale = vec2(1.0);
uniform vec2 uv_rate = vec2(0.0);   // uv units / second
uniform vec2 uv2_scale = vec2(1.0);
uniform vec2 uv2_rate = vec2(0.0);
uniform float life_max = 1.0;
uniform float alpha_ref = 0.0;
uniform float intensity = 1.0;
varying vec4 v_color;
varying vec2 v_uv2;
varying vec2 v_uv0;
varying float v_life;

void vertex() {
	if (billboard == 1) {
		mat4 mw = mat4(normalize(INV_VIEW_MATRIX[0]), normalize(INV_VIEW_MATRIX[1]), normalize(INV_VIEW_MATRIX[2]), MODEL_MATRIX[3]);
		float a = INSTANCE_CUSTOM.x;
		mw = mw * mat4(vec4(cos(a), -sin(a), 0.0, 0.0), vec4(sin(a), cos(a), 0.0, 0.0), vec4(0.0, 0.0, 1.0, 0.0), vec4(0.0, 0.0, 0.0, 1.0));
		mw = mw * mat4(vec4(length(MODEL_MATRIX[0].xyz), 0.0, 0.0, 0.0), vec4(0.0, length(MODEL_MATRIX[1].xyz), 0.0, 0.0),
				vec4(0.0, 0.0, length(MODEL_MATRIX[2].xyz), 0.0), vec4(0.0, 0.0, 0.0, 1.0));
		MODELVIEW_MATRIX = VIEW_MATRIX * mw;
		MODELVIEW_NORMAL_MATRIX = mat3(MODELVIEW_MATRIX);
	} else if (billboard == 2) {
		vec3 y = MODEL_MATRIX[1].xyz;
		float ly = max(length(y), 0.0001);
		y /= ly;
		vec3 to_cam = normalize(INV_VIEW_MATRIX[3].xyz - MODEL_MATRIX[3].xyz);
		vec3 x = normalize(cross(y, to_cam));
		vec3 z = cross(x, y);
		mat4 mw = mat4(vec4(x * length(MODEL_MATRIX[0].xyz), 0.0), vec4(y * ly, 0.0), vec4(z, 0.0), MODEL_MATRIX[3]);
		MODELVIEW_MATRIX = VIEW_MATRIX * mw;
		MODELVIEW_NORMAL_MATRIX = mat3(MODELVIEW_MATRIX);
	}
	float life = clamp(INSTANCE_CUSTOM.y, 0.0, 1.0);
	float age = life * life_max * max(INSTANCE_CUSTOM.w, 0.001);
	v_life = life;
	v_uv0 = UV;
	vec2 base_uv = UV;
	if (total_frames > 1.0) {
		float frame = mod(floor(INSTANCE_CUSTOM.z), total_frames);
		base_uv /= texdiv;
		base_uv += vec2(mod(frame, texdiv.x) / texdiv.x, floor((frame + 0.5) / texdiv.x) / texdiv.y);
	}
	vec2 off = uv_rate * age;
	vec2 off2 = uv2_rate * age;
	if (has_uv_curve) {
		vec4 k = texture(uv_curve, vec2(life, 0.5));
		off += k.rg;
		off2 += k.ba;
	}
	UV = base_uv * uv_scale + off;
	v_uv2 = v_uv0 * uv2_scale + off2;
	v_color = COLOR;
	if (linear_color) {
		v_color.rgb = pow(max(v_color.rgb, vec3(0.0)), vec3(2.2));
	}
}

void fragment() {
	vec4 c = v_color * texture(tex, UV);
	if (has_tex2) {
		c *= texture(tex2, v_uv2);
	}
	if (has_erosion) {
		float th = texture(ero_curve, vec2(v_life, 0.5)).r;
		float e = dot(texture(ero_tex, v_uv0), ero_channel);
		c.a *= clamp((e - th) / max(ero_feather, 0.01) + 0.5, 0.0, 1.0);
	}
	if (soft_dist > 0.0) {
		float d = texture(depth_tex, SCREEN_UV).r;
		vec4 vp = INV_PROJECTION_MATRIX * vec4(SCREEN_UV * 2.0 - 1.0, d, 1.0);
		float scene_z = -vp.z / vp.w;
		float frag_z = -VERTEX.z;
		c.a *= clamp((scene_z - frag_z) / soft_dist, 0.0, 1.0);
	}
	if (c.a < alpha_ref) {
		discard;
	}
	ALBEDO = c.rgb * intensity;
	ALPHA = clamp(c.a, 0.0, 1.0);
}
