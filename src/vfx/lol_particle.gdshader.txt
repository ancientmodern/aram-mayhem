shader_type spatial;
__DEFINES__
render_mode __MODES__, shadows_disabled, specular_disabled, __BLEND__;
#include "res://src/render/gamma.gdshaderinc"
// League particle material: texture x particle color, multiply texture, uv scroll / rotation / clamp, erosion
// dissolve, soft depth fade, fresnel + falloff (meshes), depth bias, billboard modes, flipbooks, distortion.
// additive materials sample raw (gamma) values to emulate League's gamma-space additive blending
uniform sampler2D tex : __TEXHINT__ filter_linear_mipmap, repeat_enable;
uniform sampler2D tex2 : __TEXHINT__ filter_linear_mipmap, repeat_enable;
uniform bool linear_color = false;   // alpha-blended materials: convert League's gamma colors to linear
uniform bool has_tex2 = false;
uniform bool tex_mask = false;      // additive avatar shells: channel-packed masks (red channel = intensity)
uniform sampler2D palette_tex : __TEXHINT__ filter_linear, repeat_disable;
uniform bool has_palette = false;   // League paletteDefinition: texture red channel picks a colour along a gradient row
uniform float palette_v = 0.5;
uniform sampler2D ero_tex : filter_linear_mipmap, repeat_enable;
uniform sampler2D ero_curve : repeat_disable, filter_linear;
uniform bool has_erosion = false;
uniform vec4 ero_channel = vec4(1.0, 0.0, 0.0, 0.0);
uniform float ero_feather = 0.1;
uniform sampler2D uv_curve : repeat_disable, filter_linear;   // rg = main uv offset, ba = mult uv offset (over life)
uniform bool has_uv_curve = false;
uniform sampler2D depth_tex : hint_depth_texture, filter_nearest;
uniform float soft_dist = 0.0;
uniform vec2 texdiv = vec2(1.0, 1.0);
uniform float total_frames = 1.0;
uniform int billboard = 1;          // 0 = particle transform, 1 = camera facing, 2 = axis (particle Y)
uniform vec2 uv_scale = vec2(1.0);
uniform vec2 uv_rate = vec2(0.0);   // uv units / second
uniform float uv_rot = 0.0;         // degrees around the quad center
uniform float uv_rot_rate = 0.0;    // degrees / second
uniform bool uv_clamp = false;
uniform bool uv_off_pack = false;  // per-particle birth uv offset packed in INSTANCE_CUSTOM.z (no flipbook)
uniform bool alpha_lock = false;   // League uvMode 2: alpha from the unscrolled uv (mask), color from the scrolled uv
uniform vec2 uv2_scale = vec2(1.0);
uniform vec2 uv2_rate = vec2(0.0);
uniform vec2 uv2_off = vec2(0.0);
uniform float uv2_rot = 0.0;
uniform bool uv2_clamp = false;
uniform vec2 uv_flip = vec2(0.0);   // 1 = flip that axis
uniform float life_max = 1.0;
uniform float alpha_ref = 0.0;
uniform float intensity = 1.0;
uniform vec4 fresnel_color = vec4(0.0);
uniform float fresnel_pow = 0.0;    // 0 = off
uniform bool rim_alpha = false;     // alpha-blended avatar shells: only the fresnel rim is opaque
uniform sampler2D falloff_tex : repeat_disable, filter_linear;
uniform bool has_falloff = false;
uniform float depth_bias = 0.0;     // meters toward the camera (League depthBiasFactors / ground layer)
uniform float distort_strength = 0.0;
#ifdef DISTORT
uniform sampler2D screen_tex : hint_screen_texture, filter_linear_mipmap, repeat_disable;
uniform sampler2D normal_tex : hint_normal, filter_linear_mipmap, repeat_enable;
#endif
instance uniform float fade = 1.0;  // linger fade-out after stop()
instance uniform vec4 tint = vec4(1.0);  // attached-mesh shells: color over life
uniform float normal_push = 0.0;    // attached-mesh shells: inflate along the normal (m)
// [vfx-runtime] League UV pipeline (meta-wiki VfxEmitterDefinitionData uvScale / VfxPrimitiveArbitraryQuad;
// ReyEngine VfxUvTransform.Cell): corner uv -> (uv - centre) * uvScale -> rotate -> + centre + translation ->
// flips -> flipbook cell ((col + u) / cols) -> texAddressModeBase
uniform bool uv_swap = false;       // VfxPrimitiveArbitraryQuad maps u = y + 0.5, v = 0.5 - x (quarter turn + mirror)
uniform int addr_mode = 0;          // texAddressModeBase: 0 wrap, 1 mirror, 2 clamp, 3 border
uniform bool lock_alpha_uv = false; // uvMode 2 LockAlpha: alpha sampled at the un-translated uv
uniform vec2 emitter_uv_rate = vec2(0.0);   // emitterUvScrollRate (system time, after the flips)
uniform float ray_offset = 0.0;     // VfxPrimitiveRay: near edge at scale.z along the axis (fraction of the length)
uniform float soft_begin = 0.0;     // softParticleParams.beginIn (m)
uniform float soft_out_begin = 0.0; // softParticleParams.beginOut (m)
uniform float soft_out_dist = 0.0;  // softParticleParams.deltaOut (m), 0 = no far fade
uniform vec4 palette_mixer = vec4(1.0, 0.0, 0.0, 0.0);   // paletteDefinition palleteSrcMixColor
uniform bool mesh_saturate = false;
varying vec2 v_uvc;                 // uv inside the flipbook cell (before the cell is applied)
varying vec2 v_uvl;                 // LockAlpha: un-translated uv inside the cell
varying flat vec2 v_cell;           // flipbook cell origin (texture uv)
varying vec4 v_color;
varying vec2 v_uv2;
varying vec2 v_uv0;
varying vec2 v_uvb;
varying float v_life;

vec2 rot_uv(vec2 uv, float deg) {
	float a = radians(deg);
	vec2 c = uv - vec2(0.5);
	return vec2(c.x * cos(a) - c.y * sin(a), c.x * sin(a) + c.y * cos(a)) + vec2(0.5);
}

void vertex() {
	if (billboard == 1) {
		mat4 mw = mat4(normalize(INV_VIEW_MATRIX[0]), normalize(INV_VIEW_MATRIX[1]), normalize(INV_VIEW_MATRIX[2]), MODEL_MATRIX[3]);
		float a = INSTANCE_CUSTOM.x;
		mw = mw * mat4(vec4(cos(a), -sin(a), 0.0, 0.0), vec4(sin(a), cos(a), 0.0, 0.0), vec4(0.0, 0.0, 1.0, 0.0), vec4(0.0, 0.0, 0.0, 1.0));
		mw = mw * mat4(vec4(length(MODEL_MATRIX[0].xyz), 0.0, 0.0, 0.0), vec4(0.0, length(MODEL_MATRIX[1].xyz), 0.0, 0.0),
				vec4(0.0, 0.0, length(MODEL_MATRIX[2].xyz), 0.0), vec4(0.0, 0.0, 0.0, 1.0));
		MODELVIEW_MATRIX = VIEW_MATRIX * mw;
		MODELVIEW_NORMAL_MATRIX = mat3(MODELVIEW_MATRIX);
	} else if (billboard == 2) {
		vec3 y = MODEL_MATRIX[1].xyz;
		float ly = max(length(y), 0.0001);
		y /= ly;
		vec3 to_cam = normalize(INV_VIEW_MATRIX[3].xyz - MODEL_MATRIX[3].xyz);
		vec3 x = normalize(cross(y, to_cam));
		vec3 z = cross(x, y);
		mat4 mw = mat4(vec4(x * length(MODEL_MATRIX[0].xyz), 0.0), vec4(y * ly, 0.0), vec4(z, 0.0), MODEL_MATRIX[3]);
		VERTEX.y += ray_offset;   // [vfx-runtime] rays start scale.z along their axis
		MODELVIEW_MATRIX = VIEW_MATRIX * mw;
		MODELVIEW_NORMAL_MATRIX = mat3(MODELVIEW_MATRIX);
	}
	if (normal_push != 0.0) {
		VERTEX += NORMAL * normal_push;
	}
	float life = clamp(INSTANCE_CUSTOM.y, 0.0, 1.0);
	float age = life * life_max * max(INSTANCE_CUSTOM.w, 0.001);
	v_life = life;
	// [vfx-runtime] corner uv of the primitive (arbitrary quads read their texture on the other diagonal)
	vec2 uvq = uv_swap ? vec2(1.0 - UV.y, 1.0 - UV.x) : UV;
	v_uv0 = uvq;
	vec2 usc = vec2(uv_scale.x == 0.0 ? 1.0 : uv_scale.x, uv_scale.y == 0.0 ? 1.0 : uv_scale.y);
	vec2 ut = (uvq - vec2(0.5)) * usc;
	float ua = radians(uv_rot + uv_rot_rate * age);
	if (ua != 0.0) {
		ut = vec2(ut.x * cos(ua) - ut.y * sin(ua), ut.x * sin(ua) + ut.y * cos(ua));
	}
	ut += vec2(0.5);
	// birth ramp (birthUVOffset + age * birthUvScrollRate): clamped to -1..1 by uvScrollClamp, else wrapped
	vec2 ramp = uv_rate * age;
	if (uv_off_pack) {
		float pk = INSTANCE_CUSTOM.z;
		ramp += vec2(floor(pk / 256.0), mod(pk, 256.0)) / 255.0;
	}
	ramp = uv_clamp ? clamp(ramp, vec2(-1.0), vec2(1.0)) : fract(ramp);
	vec2 off = ramp;
	vec2 off2 = uv2_rate * age + uv2_off;
	if (has_uv_curve) {
		vec4 k = texture(uv_curve, vec2(life, 0.5));
		off += k.rg;   // integrated particleUVScrollRate: never clamped
		off2 += k.ba;
	}
	vec2 ul = mix(ut, vec2(1.0) - ut, uv_flip);
	ut = mix(ut + off, vec2(1.0) - (ut + off), uv_flip) + emitter_uv_rate * TIME;
	v_uvl = ul;
	v_uvc = ut;
	v_cell = vec2(0.0);
	if (total_frames > 1.0) {
		float frame = mod(floor(INSTANCE_CUSTOM.z), total_frames);
		v_cell = vec2(mod(frame, texdiv.x), floor((frame + 0.5) / texdiv.x));
	}
	v_uvb = ul;
	UV = ut;
	vec2 u2 = uvq;
	if (uv2_rot != 0.0) {
		u2 = rot_uv(u2, uv2_rot);
	}
	v_uv2 = u2 * uv2_scale + off2;
	v_color = clamp(COLOR * tint, vec4(0.0), vec4(1.0));   // League particle colour is BGRA8: [0,1]
	if (linear_color) {
		v_color.rgb = pow(max(v_color.rgb, vec3(0.0)), vec3(2.2));
	}
#ifdef DEPTH_BIAS
	// pull the surface toward the camera along the view ray (same screen position, wins the depth test vs terrain)
	vec4 vp = MODELVIEW_MATRIX * vec4(VERTEX, 1.0);
	vp.xyz += normalize(-vp.xyz) * depth_bias;
	POSITION = PROJECTION_MATRIX * vp;
#endif
}

void fragment() {
	// [vfx-runtime] texAddressModeBase inside the flipbook cell, then the cell
	vec2 uc = v_uvc;
	if (addr_mode == 1) {
		uc = vec2(1.0) - abs(mod(uc, vec2(2.0)) - vec2(1.0));
	} else if (addr_mode == 2) {
		uc = clamp(uc, vec2(0.001), vec2(0.999));
	}
	vec2 uv = (v_cell + uc) / texdiv;
	vec4 tx = texture(tex, uv);
	if (addr_mode == 3 && (uc.x < 0.0 || uc.y < 0.0 || uc.x > 1.0 || uc.y > 1.0)) {
		tx = vec4(0.0);
	}
	if (tex_mask) {
		tx = vec4(vec3(tx.r), max(tx.a, tx.r));
	}
	if (has_palette) {
		// quad_ps PALETTIZE_TEXTURES: u = saturate(dot(tex, mixer)); rgb replaced, alpha = texture alpha only
		vec4 pc = texture(palette_tex, vec2(clamp(dot(tx, palette_mixer), 0.002, 0.998), palette_v));
		tx = vec4(pc.rgb, tx.a);
	}
	if (lock_alpha_uv || alpha_lock) {
		// uvMode 2 LockAlpha: colour at the scrolled uv, alpha at the uv that does not scroll
		tx.a = texture(tex, (v_cell + clamp(v_uvl, vec2(-64.0), vec2(64.0))) / texdiv).a;
	}
	// ta: texture-side alpha (texture, mult layer, erosion); va: particle-side alpha (colour, soft fade, linger)
	float ta = tx.a;
	float va = v_color.a;
	vec4 c = vec4(v_color.rgb * tx.rgb, 1.0);
	if (has_tex2) {
		vec2 u2 = uv2_clamp ? clamp(v_uv2, vec2(0.002), vec2(0.998)) : v_uv2;
		vec4 t2 = texture(tex2, u2);
		c.rgb *= t2.rgb;
		ta *= t2.a;
	}
	c.a = ta * va;
	// League alpha test (quad_ps ALPHA_TEST): clip(vertexA * texA * colourTexA - ref), before erosion / soft fade
	if (ta * va < alpha_ref) {
		discard;
	}
	if (fresnel_pow > 0.0 || has_falloff) {
		if (has_falloff) {
			c *= texture(falloff_tex, vec2(clamp(abs(dot(normalize(NORMAL), normalize(VIEW))), 0.0, 1.0), 0.5));
		}
		if (fresnel_pow > 0.0) {
			if (rim_alpha) {
				// attached-mesh shells (skinned path, not in the decoded mesh_ps): rim-only alpha
				float rim = pow(1.0 - clamp(abs(dot(normalize(NORMAL), normalize(VIEW))), 0.0, 1.0), fresnel_pow);
				c.rgb += fresnel_color.rgb * rim;
				c.a *= rim;
			} else {
				// mesh_vs/mesh_ps [exact]: F = 1 - pow(saturate(N.V), fresnel); rgb += fresnelColor * F * alpha
				float ndv = clamp(dot(normalize(NORMAL), normalize(VIEW)), 0.0, 1.0);
				c.rgb += fresnel_color.rgb * (1.0 - pow(ndv, fresnel_pow)) * c.a;
			}
		}
		ta = c.a / max(va, 0.0001);
	}
	if (has_erosion) {
		float th = texture(ero_curve, vec2(v_life, 0.5)).r;
		float e = dot(texture(ero_tex, v_uv0), ero_channel);
		float ek = clamp((e - th) / max(ero_feather, 0.01) + 0.5, 0.0, 1.0);
		c.a *= ek;
		ta *= ek;
	}
	float pf = fade;
	if (soft_dist > 0.0) {
		float d = texture(depth_tex, SCREEN_UV).r;
		vec4 vp = INV_PROJECTION_MATRIX * vec4(SCREEN_UV * 2.0 - 1.0, d, 1.0);
		float scene_z = -vp.z / vp.w;
		float frag_z = -VERTEX.z;
		float dz = scene_z - frag_z;
		float sf = smoothstep(soft_begin, soft_begin + soft_dist, dz);
		if (soft_out_dist > 0.0) {
			sf -= smoothstep(soft_out_begin, soft_out_begin + soft_out_dist, dz);   // optional far fade [exact]
		}
		pf *= clamp(sf, 0.0, 1.0);
	}
	c.a *= pf;
#ifdef PREMUL_ADD
	// League blendMode 0 (Add, the default): the particle colour is premultiplied by its alpha on the CPU and
	// drawn ONE,ONE - texture alpha only feeds the alpha test (meta-wiki blendMode; ReyEngine vfx-blend-modes)
	c.rgb *= va * pf;
	c.a = 1.0;
#endif
	if (mesh_saturate) {
		c.rgb = clamp(c.rgb, vec3(0.0), vec3(1.0));   // mesh_ps saturates its output colour [exact]
	}
#ifdef DISTORT
	{
		// distortion_ps [exact]: suv = screenUV + (2n - 1) * DistortionPower * colourTex.a; rgb = bg * tex.rgb * colour.rgb;
		// a = normalMap.a. The soft quad border only hides Godot's screen copy (it lacks the transparent pass).
		vec4 nt = texture(normal_tex, uv);
		vec2 n = nt.rg * 2.0 - 1.0;
		vec2 bd = smoothstep(vec2(0.0), vec2(0.12), v_uv0) * smoothstep(vec2(0.0), vec2(0.12), vec2(1.0) - v_uv0);
		vec2 suv = SCREEN_UV + n * distort_strength;
		ALBEDO = textureLod(screen_tex, suv, 0.0).rgb * tx.rgb * clamp(v_color.rgb, vec3(0.0), vec3(1.0));
		ALPHA = clamp(nt.a * bd.x * bd.y, 0.0, 1.0);
	}
#elif defined(SOLID)
	// League blendMode 3: solid meshes (rocks, rockets, relic icons). League draws them unlit with lighting
	// painted into the texture; a gentle fixed key light keeps dark ones (Malphite's shard) readable as 3D shapes
	float key = clamp(dot(normalize(NORMAL), normalize(vec3(-0.35, 0.8, 0.5))), 0.0, 1.0);
	ALBEDO = to_out(c.rgb * (0.8 + 0.45 * key));
	ALPHA = clamp(c.a, 0.0, 1.0);
#elif defined(RAW_TEX)
	// raw (gamma) texture x League gamma colour: already display-space; the 0.85 damping only compensates
	// the linear-buffer emulation
	ALBEDO = c.rgb * (gamma_out > 0.5 ? 1.0 : intensity);
	ALPHA = clamp(c.a, 0.0, 1.0);
#else
	ALBEDO = to_out(c.rgb * intensity);
	ALPHA = clamp(c.a, 0.0, 1.0);
#endif
}
